cmake_minimum_required(VERSION 3.21)
project(LearningCStudio VERSION 1.0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network)
qt_standard_project_setup()
if(MSVC)
  add_compile_options(/utf-8 /W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(studio_core protocol.h tcpclient.h tcpclient.cpp httpclient.h httpclient.cpp benchmark.h benchmark.cpp)
target_link_libraries(studio_core PUBLIC Qt6::Network)
target_include_directories(studio_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(studio_ui mainwindow.h mainwindow.cpp)
target_link_libraries(studio_ui PUBLIC studio_core Qt6::Widgets)
qt_add_executable(learningc_studio main.cpp)
target_link_libraries(learningc_studio PRIVATE studio_ui)
set_target_properties(learningc_studio PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  foreach(name IN ITEMS http benchmark tcp)
    qt_add_executable(test_${name} test_${name}.cpp)
    target_link_libraries(test_${name} PRIVATE studio_core Qt6::Test)
    add_test(NAME test_${name} COMMAND test_${name} -o "${CMAKE_BINARY_DIR}/test_${name}-results.txt,txt")
    set_tests_properties(test_${name} PROPERTIES TIMEOUT 90)
  endforeach()
  qt_add_executable(test_ui test_ui.cpp)
  target_link_libraries(test_ui PRIVATE studio_ui Qt6::Test)
  if(WIN32)
    add_test(NAME test_ui COMMAND test_ui -platform windows -o "${CMAKE_BINARY_DIR}/test_ui-results.txt,txt")
  else()
    add_test(NAME test_ui COMMAND test_ui -platform offscreen -o "${CMAKE_BINARY_DIR}/test_ui-results.txt,txt")
  endif()
  set_tests_properties(test_ui PROPERTIES TIMEOUT 90)
endif()
install(TARGETS learningc_studio RUNTIME DESTINATION bin BUNDLE DESTINATION .)
